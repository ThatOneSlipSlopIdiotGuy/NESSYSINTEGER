cmake_minimum_required(VERSION 3.12)
project(NESSYSINTEGER C)
set(CMAKE_C_STANDARD 99)
set(CORE src/cpu.c src/ppu.c src/apu.c src/cart.c src/nes.c src/boards.c src/boards_konami.c src/boards_misc.c src/boards_mmc5.c src/ppuview.c)
if(MSVC)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
  add_executable(NESSYSINTEGER WIN32 ${CORE} win/main.c)
  target_link_libraries(NESSYSINTEGER gdi32 user32 winmm comdlg32 shell32)
endif()
add_executable(nessys_cli ${CORE} tools/nessys_cli.c)
add_executable(dma_trace ${CORE} tests/dma_trace.c)
add_executable(accuracycoin_run ${CORE} tools/accuracycoin_run.c)
add_executable(mapper_test ${CORE} tests/mapper_test.c)
if(NOT WIN32)
  target_link_libraries(nessys_cli m)
  target_link_libraries(dma_trace m)
  target_link_libraries(accuracycoin_run m)
  target_link_libraries(mapper_test m)
endif()
